# Ring Nebula Renderer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** M57 becomes an explorable, true-3D planetary nebula in Lyra that matches its JWST and Hubble depictions, and `planetary_nebula` becomes a supported deep-sky type.

**Architecture:** Two gated parameters (`waist`, `halo`) are added to the shared nebula engine's gas-shape function, in GLSL and in its JavaScript twin. Both are off at 0, which is proven by node tests and a pixel-identity check against M42 and M31. An M57 look is tuned in the nebula lab, stored as the `planetary_nebula` preset, and drafted into Lyra through the existing pipeline. It ships to app-preview first; live waits for the user.

**Tech Stack:** plain-script WebGL engine (`nebula-core.js`, three r137 via 8frame), A-Frame 1.3 app, Node 24 `node:test`, the draft/promote tools.

**Spec:** `docs/superpowers/specs/2026-10-07-ring-nebula-renderer-design.md`

## Global Constraints

- The engine lives in a DIFFERENT repo: `star-gallery/nebula-shared/nebula-core.js` (nested repo `star-gallery/`, LF line endings). The live app and app-preview both load it, so it reaches live the moment `star-gallery` is pushed.
- New parameters default to `0`, and every new code path is gated `> 0`. Orion's M42 and Andromeda's M31 must render pixel-identically before and after (spec §3.4). Any difference stops the work.
- No colour code is added to the engine (spec §3.2). If the look cannot be reached with the existing ramp, stop and report to the user (Task 4).
- M56 stays `layer: "none"`.

- Run all tool tests with `node --test "tools/test/*.test.mjs"` (the glob, not the directory).
- Push with the gh credential helper (the stored git credential returns HTTP 500s):

- Source repo pushes use explicit refspecs: `main:refs/heads/source/main`. Never push the source repo's `main` to `origin/main`: that branch is the live site.
- Rollout (a): app-preview and a QR first. **Do not promote to live in this plan.** Live waits for the user's device check.

```js
// tools/test/helpers/load-nebula-core.mjs - the shared nebula engine, in node.
//
// nebula-core.js is a browser script: an IIFE over `window` that reads window.THREE. The shape
// maths only needs Vector3, so the rest of THREE is a permissive stub - enough for the file's
// top-level `new THREE.Quaternion()` and friends to construct something.
import {createRequire} from 'node:module'
import path from 'node:path'

const require = createRequire(import.meta.url)
export const ENGINE = 'star-gallery/nebula-shared/nebula-core.js'

class Vector3 {
  constructor(x = 0, y = 0, z = 0) { this.x = x; this.y = y; this.z = z }
  set(x, y, z) { this.x = x; this.y = y; this.z = z; return this }
  multiplyScalar(s) { this.x *= s; this.y *= s; this.z *= s; return this }
}

export function loadNebulaCore(file = ENGINE) {
  const THREE = new Proxy({Vector3}, {get: (t, k) => t[k] || class {}})
  const saved = globalThis.window
  globalThis.window = {THREE, performance: globalThis.performance}
  const abs = path.resolve(file)
  delete require.cache[abs]
  try {
    require(abs)
    return globalThis.window.NebulaCore
  } finally {
    globalThis.window = saved
  }
}
```

- [ ] **Step 1: The JavaScript twin.** Replace the body of `shapeMulJS`'s `hollow` block, and add a `waist` block after `bipolar`:

```js
  function shapeMulJS(x, y, z, P) {
    var m = 1
    if (P.hollow > 0) {
      var R = envRadii(P), qx = x / R.x, qy = y / R.y, qz = z / R.z
      var e = qx * qx + qy * qy + qz * qz
      if (e > 1) return 0
      var r = Math.sqrt(e), wl = (r - P.shellR) * P.shellK
      var wall = Math.exp(-wl * wl)
      // Faint gas outside the wall, fading out by haloR: a planetary nebula's outer halo.
      if (P.halo > 0) {
        wall += P.halo * sstep(P.shellR, P.shellR + 1 / P.shellK, r) *
          (1 - sstep(P.shellR, P.haloR, r)) * (1 - e)
      }
      m *= (1 - P.hollow) + P.hollow * wall
    }
    if (P.bipolar > 0) {
      var ax = dirFrom(P.lobeAz, P.lobeEl)
      var rx = x - P.sunX, ry = y - P.sunY, rz = z - P.sunZ
      var len = Math.sqrt(rx * rx + ry * ry + rz * rz) || 1e-6
      var c = (rx * ax.x + ry * ax.y + rz * ax.z) / len
      var lobe = Math.max(0, Math.min(1, c * c * (1 + P.lobeBias * c)))
      lobe = lobe + (lobe * lobe - lobe) * P.lobeSharp
      m *= 1 + (lobe - 1) * P.bipolar
    }
    // The inverse of bipolar: dense around the equator, open at the poles - the barrel a
    // planetary nebula like M57 is, which reads as a ring seen near its axis.
    if (P.waist > 0) {
      var wa = dirFrom(P.lobeAz, P.lobeEl)
      var bx = x - P.sunX, by = y - P.sunY, bz = z - P.sunZ
      var blen = Math.sqrt(bx * bx + by * by + bz * bz) || 1e-6
      var bc = (bx * wa.x + by * wa.y + bz * wa.z) / blen
      var belt = Math.pow(Math.max(0, 1 - bc * bc), 1 + P.waistSharp * 6)
      m *= 1 + (belt - 1) * P.waist
    }
    return m
  }
```

```html
// Renders M42 and M31 with the committed engine (A) and the working copy (B), each with
// Math.random seeded identically and every time-driven uniform pinned, then compares the bytes.
// Any difference means the new parameters are not truly off at their defaults.
function load(src) {
  return new Promise(function (res, rej) {
    var s = document.createElement('script'); s.src = src + '?t=' + Date.now()
    s.onload = function () { var c = window.NebulaCore; window.NebulaCore = undefined; res(c) }
    s.onerror = rej; document.body.appendChild(s)
  })
}
function seeded(seed) {   // mulberry32
  return function () {
    seed |= 0; seed = seed + 0x6D2B79F5 | 0
    var t = Math.imul(seed ^ seed >>> 15, 1 | seed)
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t
    return ((t ^ t >>> 14) >>> 0) / 4294967296
  }
}
var SIZE = 512
var renderer = new THREE.WebGLRenderer({antialias: false, preserveDrawingBuffer: true})
renderer.setSize(SIZE, SIZE, false); renderer.setPixelRatio(1); renderer.setClearColor(0x000000, 1)
document.body.appendChild(renderer.domElement)

function shoot(core, preset) {
  var rnd = Math.random; Math.random = seeded(1234)
  try {
    var P = Object.assign(core.params(), preset), state = {}
    var out = core.build(P, preset.mode, state, renderer.capabilities.isWebGL2)

}
```
